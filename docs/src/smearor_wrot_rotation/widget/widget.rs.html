42    /// Transform input coordinates based on current rotation
43    pub fn input_transform(&self, x: f64, y: f64) -> (f64, f64) {
44        let rotation = self.rotation();
45        if rotation == 0.0 {
46            (x, y)
47        } else {
48            // Get widget dimensions
49            let width = self.width() as f64;
50            let height = self.height() as f64;
51
52            // Translate to center
53            let cx = width / 2.0;
54            let cy = height / 2.0;
55
56            let dx = x - cx;
57            let dy = y - cy;
58
59            // Rotate in reverse direction to compensate for widget rotation
60            let radians = (-rotation).to_radians() as f64;
61            let cos = radians.cos();
62            let sin = radians.sin();
63
64            let rotated_x = dx * cos - dy * sin;
65            let rotated_y = dx * sin + dy * cos;
66
67            // Translate back
68            (rotated_x + cx, rotated_y + cy)
69        }
70    }

93    pub fn set_rotation_with_animation(&self, new_rotation: f64) {
94        let imp = self.imp();
95
96        // Check if animations are disabled
97        if !imp.animations_enabled.get() {
98            // If animations are disabled, set rotation immediately
99            let layout = match self.layout_manager() {
100                Some(lm) => lm,
101                None => {
102                    error!("Failed to get layout manager in set_rotation_with_animation");
103                    return;
104                }
105            };
106
107            let layout = match layout.downcast::<RotatedLayout>() {
108                Ok(rl) => rl,
109                Err(_) => {
110                    error!("Layout manager is not RotatedLayout in set_rotation_with_animation");
111                    return;
112                }
113            };
114
115            layout.imp().rotation.set(new_rotation as f32);
116            self.queue_allocate();
117            return;
118        }
119
120        let layout = match self.layout_manager() {
121            Some(lm) => lm,
122            None => {
123                error!("Failed to get layout manager in set_rotation_with_animation");
124                return;
125            }
126        };
127
128        let layout = match layout.downcast::<RotatedLayout>() {
129            Ok(rl) => rl,
130            Err(_) => {
131                error!("Layout manager is not RotatedLayout in set_rotation_with_animation");
132                return;
133            }
134        };
135
136        let current_rotation = layout.imp().rotation.get();
137        let animation_speed_ms = imp.animation_speed.get();
138
139        // Create rotation zoom animation with three phases
140        // 0-33%: Rotation and zoom out (scale 1.0 -> 0.9)
141        // 33-66%: Rotation without zoom (scale 0.9 -> 0.9)
142        // 66-100%: Rotation and zoom in (scale 0.9 -> 1.0)
143        let mut animation = RotationZoomAnimation::new(
144            current_rotation as f64,
145            new_rotation,
146            1.0,
147            0.8,
148            1.0,
149            Duration::from_millis(animation_speed_ms),
150            EasingFunction::EaseInOut,
151        );
152        animation.start();
153        *imp.rotation_zoom_animation.borrow_mut() = Some(animation);
154
155        self.add_tick_callback(move |widget, _frame_clock| {
156            let imp = widget.imp();
157            if let Some(ref mut anim) = *imp.rotation_zoom_animation.borrow_mut() {
158                let (current_rotation, current_scale): (f64, f64) = match anim.get_current_values_with_phases() {
159                    Some(values) => values,
160                    None => return glib::ControlFlow::Break,
161                };
162
163                let layout = match widget.layout_manager() {
164                    Some(lm) => lm,
165                    None => return glib::ControlFlow::Continue,
166                };
167
168                let layout = match layout.downcast::<RotatedLayout>() {
169                    Ok(rl) => rl,
170                    Err(_) => return glib::ControlFlow::Continue,
171                };
172
173                // Apply rotation
174                layout.imp().rotation.set(current_rotation as f32);
175
176                // Apply scale using transform matrix instead of CSS
177                layout.imp().scale.set(current_scale as f32);
178
179                widget.queue_allocate();
180
181                if anim.is_complete() {
182                    // Reset scale to 1.0 after animation completes
183                    layout.imp().scale.set(1.0);
184                    glib::ControlFlow::Break
185                } else {
186                    glib::ControlFlow::Continue
187                }
188            } else {
189                glib::ControlFlow::Break
190            }
191        });
192    }

194    pub fn set_child(&self, child: Option<&impl IsA<gtk4::Widget>>) {
195        let mut self_child = self.imp().child.borrow_mut();
196        if let Some(old_child) = self_child.take() {
197            old_child.unparent();
198        }
199        if let Some(new_child) = child {
200            let widget_clone_width = self.clone();
201            let widget_clone_height = self.clone();
202            new_child.connect_notify_local(Some("width-request"), move |_child, _param| {
203                widget_clone_width.queue_allocate();
204            });
205            new_child.connect_notify_local(Some("height-request"), move |_child, _param| {
206                widget_clone_height.queue_allocate();
207            });
208            new_child.set_parent(self);
209            *self_child = Some(new_child.clone().upcast());
210        }
211    }
