22pub struct RotatedLayoutImpl {
23    pub rotation: Cell<RotationRad>,
24    pub scale: Cell<ScaleFactor>,
25    /// When `true`, the child is allocated its measured natural size.
26    /// When `false`, it is resized to the largest rectangle with its
27    /// natural aspect ratio whose rotated bounding box fits inside the
28    /// allocation (`unrotated_fit` - a one-sided condition that is never
29    /// singular).
30    pub natural_child_size: Cell<bool>,
31    /// When `true` (default), `measure` requests the rotated bounding
32    /// box, so the allocation always contains the rotated child and the
33    /// viewport grows with rotation. When `false`, the request stays
34    /// unrotated; `allocate` still centers the child, so it overflows
35    /// the fixed viewport symmetrically. Combine with
36    /// `natural_child_size` to keep the child at its natural size, or
37    /// leave it disabled to shrink the child via `unrotated_fit` so it
38    /// still fits rotated inside the fixed viewport.
39    pub measure_rotated_bounds: Cell<bool>,
40}
41
42impl Default for RotatedLayoutImpl {
43    fn default() -> Self {
44        Self {
45            rotation: Cell::new(RotationRad::ZERO),
46            scale: Cell::new(ScaleFactor::ONE),
47            natural_child_size: Cell::new(false),
48            measure_rotated_bounds: Cell::new(true),
49        }
50    }
51}

65    pub const MEASURE_ROTATED_BOUNDS: &str = "measure-rotated-bounds";

68impl ObjectImpl for RotatedLayoutImpl {
69    /// The layout's state is stored in `Cell` newtypes that do not
70    /// implement `glib::property::Property`, so the properties are
71    /// registered manually as plain `f64`/`bool` values mapped onto the
72    /// cells. Registered properties are required for `notify::rotation`
73    /// to reach consumers (e.g. the floating board's re-centering).
74    fn properties() -> &'static [glib::ParamSpec] {
75        static PROPERTIES: LazyLock<Vec<glib::ParamSpec>> = LazyLock::new(|| {
76            vec![
77                glib::ParamSpecDouble::builder(prop::ROTATION).build(),
78                glib::ParamSpecDouble::builder(prop::SCALE).build(),
79                glib::ParamSpecBoolean::builder(prop::NATURAL_CHILD_SIZE).build(),
80                glib::ParamSpecBoolean::builder(prop::MEASURE_ROTATED_BOUNDS).build(),
81            ]
82        });
83        PROPERTIES.as_slice()
84    }
85
86    fn property(&self, _id: usize, pspec: &glib::ParamSpec) -> glib::Value {
87        match pspec.name() {
88            prop::ROTATION => self.rotation.get().as_f64().to_value(),
89            prop::SCALE => self.scale.get().as_f64().to_value(),
90            prop::NATURAL_CHILD_SIZE => self.natural_child_size.get().to_value(),
91            prop::MEASURE_ROTATED_BOUNDS => self.measure_rotated_bounds.get().to_value(),
92            name => {
93                warn!(%name, "unknown RotatedLayout property");
94                pspec.default_value().to_owned()
95            }
96        }
97    }
98
99    fn set_property(&self, _id: usize, value: &glib::Value, pspec: &glib::ParamSpec) {
100        match pspec.name() {
101            prop::ROTATION => {
102                if let Ok(rotation) = value.get::<f64>() {
103                    self.rotation.set(RotationRad::new(rotation));
104                }
105            }
106            prop::SCALE => {
107                if let Ok(scale) = value.get::<f64>() {
108                    self.scale.set(ScaleFactor::new(scale));
109                }
110            }
111            prop::NATURAL_CHILD_SIZE => {
112                if let Ok(enabled) = value.get::<bool>() {
113                    self.natural_child_size.set(enabled);
114                }
115            }
116            prop::MEASURE_ROTATED_BOUNDS => {
117                if let Ok(enabled) = value.get::<bool>() {
118                    self.measure_rotated_bounds.set(enabled);
119                }
120            }
121            name => warn!(%name, "unknown RotatedLayout property"),
122        }
123    }
124}

126impl LayoutManagerImpl for RotatedLayoutImpl {
127    fn allocate(&self, widget: &gtk4::Widget, width: i32, height: i32, baseline: i32) {
128        let Some(c) = widget.first_child().filter(|c| c.should_layout()) else {
129            return;
130        };
131        let rotation = self.rotation.get();
132        let allocation = Size::new(width, height);
133        let center = allocation.to_f32().half();
134        let (_, child_nat) = c.measure_size();
135        let target_size: Size<i32> = if self.natural_child_size.get() {
136            // The child keeps its measured natural size: `measure`
137            // reports the rotated bounding box, so the allocation
138            // always contains the rotated child.
139            child_nat
140        } else {
141            // The child keeps its natural aspect ratio and is resized
142            // to the largest rectangle whose rotated bounding box fits
143            // inside the allocation. `unrotated_fit` is a one-sided
144            // condition, so it is never singular - the previous exact
145            // `unrotated_bounds` solve forced a bounds-fill aspect
146            // onto the child, letterboxing aspect-fixed content such
147            // as a wide image down to an invisible sliver.
148            let nat_aspect = child_nat.to_f64().aspect().unwrap_or(AspectRatio::ONE);
149            allocation.to_f64().unrotated_fit(nat_aspect, rotation).map(|v| v as i32)
150        };
151
152        let transform: Transform = CenteredTransform::new(center, target_size.to_f32(), rotation, self.scale.get()).into();
153
154        trace!("size {target_size} baseline {baseline} transform {transform:?}");
155        c.allocate(target_size.x, target_size.y, baseline, Some(transform));
156    }
157
158    fn measure(&self, widget: &gtk4::Widget, orientation: Orientation, _for_size: i32) -> (i32, i32, i32, i32) {
159        let rotation = self.rotation.get();
160
161        let Some(child) = widget.first_child() else {
162            debug!("No child");
163            return (0, 0, -1, -1);
164        };
165        let (child_min, child_nat) = child.measure_size();
166        trace!("child_min {child_min} child_nat {child_nat}");
167
168        // `measure_rotated_bounds` requests the rotated bounding box, so
169        // the allocation always contains the rotated child; disabled,
170        // the request stays unrotated and the child overflows its
171        // allocation symmetrically (`allocate` centers either way).
172        let bounds = |size: Size<i32>| {
173            if self.measure_rotated_bounds.get() {
174                let rotated = size.to_f64().rotated_bounds(rotation).to_i32_ceil();
175                Size::new(rotated.x, rotated.y)
176            } else {
177                size
178            }
179        };
180        let (min, nat) = (bounds(child_min), bounds(child_nat));
181        if orientation == Orientation::Horizontal {
182            (min.x, nat.x, -1, -1)
183        } else {
184            (min.y, nat.y, -1, -1)
185        }
186    }
