19#[derive(Properties)]
20#[properties(wrapper_type = RotatedLayout)]
21pub struct RotatedLayoutImpl {
22    #[properties(get, set, notify)]
23    pub rotation: Cell<RotationRad>,
24    #[properties(get, set, notify)]
25    pub scale: Cell<ScaleFactor>,
26    /// When `true`, the child is allocated its measured natural size instead
27    /// of solving which rotated child size reproduces the allocation. The
28    /// inverse solve is ill-conditioned near 45 degrees (and symmetric bands),
29    /// where it shrinks non-square children into a square. Defaults to
30    /// `false` to preserve the legacy allocation behavior.
31    #[properties(get, set, notify)]
32    pub natural_child_size: Cell<bool>,
33}
34
35impl Default for RotatedLayoutImpl {
36    fn default() -> Self {
37        Self {
38            rotation: Cell::new(RotationRad::ZERO),
39            scale: Cell::new(ScaleFactor::ONE),
40            natural_child_size: Cell::new(false),
41        }
42    }
43}

55impl LayoutManagerImpl for RotatedLayoutImpl {
56    fn allocate(&self, widget: &gtk4::Widget, width: i32, height: i32, baseline: i32) {
57        let Some(c) = widget.first_child().filter(|c| c.should_layout()) else {
58            return;
59        };
60        let rotation = self.rotation.get();
61        let angle_deg = rotation.to_degrees_f32();
62        let current_scale = self.scale.get().as_f32();
63        let (sin, cos) = rotation.sin_cos();
64        let abs_cos = cos.abs();
65        let abs_sin = sin.abs();
66        // `denom` is `cos(2θ)` and vanishes at 45° + k·90° - the
67        // matrix singularity where no rotated child size can
68        // reproduce a non-square allocation and the inverse
69        // solve diverges. Blend quadratically from the solved
70        // size at the band edge (`|denom| = 0.9`, continuous
71        // with the pure solve outside) down to the square fit
72        // at the singularity; a hard switch jumped the child
73        // size whenever `|denom|` crossed the threshold.
74        let denom = abs_cos * abs_cos - abs_sin * abs_sin;
75        let square_fit = width as f64 / (abs_cos + abs_sin);
76        let blend = (denom.abs() / 0.9).min(1.0);
77        let blend = blend * blend;
78        // At the singularity `unrotated_bounds` holds +-inf/NaN -
79        // `blend == 0.0` takes the square-fit branch and never reads it.
80        let solved = Size::new(width, height).to_f64().unrotated_bounds(rotation);
81        let center = Position::<f32>::new(width as f32 / 2.0, height as f32 / 2.0);
82        let target_size: Size<i32> = if self.natural_child_size.get() {
83            // The child keeps its measured natural size: `measure`
84            // reports the rotated bounding box, so the allocation
85            // always contains the rotated child.
86            let (_, child_nat_w, _, _) = c.measure(Orientation::Horizontal, -1);
87            let (_, child_nat_h, _, _) = c.measure(Orientation::Vertical, -1);
88            Size::new(child_nat_w, child_nat_h)
89        } else if blend == 0.0 {
90            let s = square_fit as i32;
91            Size::new(s, s)
92        } else {
93            solved.map(|v| (square_fit + (v.max(0.0) - square_fit) * blend) as i32)
94        };
95
96        let child_center = target_size.map(|v| v as f32 / -2.0);
97        let transform = Transform::new()
98            .translate(&center.into())
99            .rotate(angle_deg)
100            .scale(current_scale, current_scale)
101            .translate(&child_center.into());
102
103        trace!("size {target_size} baseline {baseline} transform {transform:?}");
104        c.allocate(target_size.x, target_size.y, baseline, Some(transform));
105    }
106
107    fn measure(&self, widget: &gtk4::Widget, orientation: Orientation, _for_size: i32) -> (i32, i32, i32, i32) {
108        let rotation = self.rotation.get();
109
110        if let Some(child) = widget.first_child() {
111            let (child_min_w, child_nat_w, _, _) = child.measure(Orientation::Horizontal, -1);
112            let (child_min_h, child_nat_h, _, _) = child.measure(Orientation::Vertical, -1);
113            trace!("child_min_w {child_min_w} child_nat_w {child_nat_w} child_min_h {child_min_h} child_nat_h {child_nat_h}");
114
115            // Rotated bounding box of the child's natural and minimum size.
116            let min = Size::new(child_min_w, child_min_h).to_f64().rotated_bounds(rotation);
117            let nat = Size::new(child_nat_w, child_nat_h).to_f64().rotated_bounds(rotation);
118            if orientation == Orientation::Horizontal {
119                (min.to_i32_ceil().x, nat.to_i32_ceil().x, -1, -1)
120            } else {
121                (min.to_i32_ceil().y, nat.to_i32_ceil().y, -1, -1)
122            }
123        } else {
124            debug!("No child");
125            (0, 0, -1, -1)
126        }
127    }
