45    fn allocate(&self, widget: &gtk4::Widget, width: i32, height: i32, baseline: i32) {
46        let mut child = widget.first_child();
47        let angle_deg = self.rotation.get();
48        let current_scale = self.scale.get();
49        let angle_rad = angle_deg.to_radians();
50        let abs_cos = angle_rad.cos().abs();
51        let abs_sin = angle_rad.sin().abs();
52        while let Some(ref c) = child {
53            if c.should_layout() {
54                let denom = abs_cos * abs_cos - abs_sin * abs_sin;
55                let (target_child_w, target_child_h) = if denom.abs() < 0.9 {
56                    // Sonderfall 45°, 135° etc. (Matrix singulär)
57                    // Hier vereinfacht: Kind quadratisch einpassen
58                    let s = (width as f32 / (abs_cos + abs_sin)) as i32;
59                    (s, s)
60                } else {
61                    let w_c = (width as f32 * abs_cos - height as f32 * abs_sin) / denom;
62                    let h_c = (height as f32 * abs_cos - width as f32 * abs_sin) / denom;
63                    (w_c as i32, h_c as i32)
64                };
65
66                // let (_, child_nat_w, _, _) = c.measure(Orientation::Horizontal, -1);
67                // let (_, child_nat_h, _, _) = c.measure(Orientation::Vertical, -1);
68                // trace!("child_nat_w {child_nat_w} child_nat_h {child_nat_h}");
69                let transform = Transform::new()
70                    .translate(&Point::new(width as f32 / 2.0, height as f32 / 2.0))
71                    .rotate(angle_deg)
72                    .scale(current_scale, current_scale)
73                    // .translate(&Point::new(child_nat_w as f32 / -2.0, child_nat_h as f32 / -2.0));
74                    .translate(&Point::new(target_child_w as f32 / -2.0, target_child_h as f32 / -2.0));
75
76                trace!("width {target_child_w} height {target_child_h} baseline {baseline} transform {transform:?}");
77                // c.allocate(width, height, baseline, Some(transform));
78                c.allocate(target_child_w, target_child_h, baseline, Some(transform));
79            }
80            child = c.next_sibling();
81        }
82    }

84    fn measure(&self, widget: &gtk4::Widget, orientation: Orientation, _for_size: i32) -> (i32, i32, i32, i32) {
85        let angle_deg = self.rotation.get();
86        let angle_rad = angle_deg.to_radians();
87        let abs_cos = angle_rad.cos().abs();
88        let abs_sin = angle_rad.sin().abs();
89
90        if let Some(child) = widget.first_child() {
91            let (child_min_w, child_nat_w, _, _) = child.measure(Orientation::Horizontal, -1);
92            let (child_min_h, child_nat_h, _, _) = child.measure(Orientation::Vertical, -1);
93            trace!("child_min_w {child_min_w} child_nat_w {child_nat_w} child_min_h {child_min_h} child_nat_h {child_nat_h}");
94
95            if orientation == Orientation::Horizontal {
96                // Calculate bounding box width
97                let min = (child_min_w as f32 * abs_cos + child_min_h as f32 * abs_sin).ceil() as i32;
98                let nat = (child_nat_w as f32 * abs_cos + child_nat_h as f32 * abs_sin).ceil() as i32;
99                (min, nat, -1, -1)
100            } else {
101                // Calculate bounding box height
102                let min = (child_min_w as f32 * abs_sin + child_min_h as f32 * abs_cos).ceil() as i32;
103                let nat = (child_nat_w as f32 * abs_sin + child_nat_h as f32 * abs_cos).ceil() as i32;
104                (min, nat, -1, -1)
105            }
106        } else {
107            debug!("No child");
108            (0, 0, -1, -1)
109        }
110    }
