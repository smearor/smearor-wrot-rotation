23pub struct RotatedLayoutImpl {
24    pub rotation: Cell<RotationRad>,
25    pub scale: Cell<ScaleFactor>,
26    /// When `true`, the child is allocated its measured natural size.
27    /// When `false`, it is resized to the largest rectangle with its
28    /// natural aspect ratio whose rotated bounding box fits inside the
29    /// allocation (`unrotated_fit` - a one-sided condition that is never
30    /// singular).
31    pub natural_child_size: Cell<bool>,
32    /// When `true` (default), `measure` requests the rotated bounding
33    /// box, so the allocation always contains the rotated child and the
34    /// viewport grows with rotation. When `false`, the request stays
35    /// unrotated; `allocate` still centers the child, so it overflows
36    /// the fixed viewport symmetrically. Combine with
37    /// `natural_child_size` to keep the child at its natural size, or
38    /// leave it disabled to shrink the child via `unrotated_fit` so it
39    /// still fits rotated inside the fixed viewport.
40    pub measure_rotated_bounds: Cell<bool>,
41    /// When `true`, the child is stretched to fill the whole allocation:
42    /// it is allocated the size whose rotated bounding box exactly equals
43    /// the allocation (`unrotated_bounds`). Takes precedence over
44    /// `natural_child_size`. Where the exact solve is infeasible the
45    /// `fill_strategy` fallback resolves the allocation.
46    pub fill: Cell<bool>,
47    /// How `fill` resolves rotation angles where the exact
48    /// `unrotated_bounds` solve is infeasible - see [`FillStrategy`].
49    pub fill_strategy: Cell<FillStrategy>,
50}
51

56            scale: Cell::new(ScaleFactor::ONE),

58            measure_rotated_bounds: Cell::new(true),

82impl ObjectImpl for RotatedLayoutImpl {
83    /// The layout's state is stored in `Cell` newtypes that do not
84    /// implement `glib::property::Property`, so the properties are
85    /// registered manually as plain `f64`/`bool` values mapped onto the
86    /// cells. Registered properties are required for `notify::rotation`
87    /// to reach consumers (e.g. the floating board's re-centering).
88    fn properties() -> &'static [glib::ParamSpec] {
89        static PROPERTIES: LazyLock<Vec<glib::ParamSpec>> = LazyLock::new(|| {
90            vec![
91                glib::ParamSpecDouble::builder(prop::ROTATION).build(),
92                glib::ParamSpecDouble::builder(prop::SCALE).build(),
93                glib::ParamSpecBoolean::builder(prop::NATURAL_CHILD_SIZE).build(),
94                glib::ParamSpecBoolean::builder(prop::MEASURE_ROTATED_BOUNDS).build(),
95                glib::ParamSpecBoolean::builder(prop::FILL).build(),
96                glib::ParamSpecEnum::builder::<FillStrategy>(prop::FILL_STRATEGY).build(),
97            ]
98        });
99        PROPERTIES.as_slice()
100    }

102    fn property(&self, _id: usize, pspec: &glib::ParamSpec) -> glib::Value {
103        match pspec.name() {
104            prop::ROTATION => self.rotation.get().as_f64().to_value(),
105            prop::SCALE => self.scale.get().as_f64().to_value(),
106            prop::NATURAL_CHILD_SIZE => self.natural_child_size.get().to_value(),
107            prop::MEASURE_ROTATED_BOUNDS => self.measure_rotated_bounds.get().to_value(),
108            prop::FILL => self.fill.get().to_value(),
109            prop::FILL_STRATEGY => self.fill_strategy.get().to_value(),
110            name => {
111                warn!(%name, "unknown RotatedLayout property");
112                pspec.default_value().to_owned()
113            }
114        }
115    }
116
117    fn set_property(&self, _id: usize, value: &glib::Value, pspec: &glib::ParamSpec) {
118        match pspec.name() {
119            prop::ROTATION => {
120                if let Ok(rotation) = value.get::<f64>() {
121                    self.rotation.set(RotationRad::new(rotation));
122                }
123            }
124            prop::SCALE => {
125                if let Ok(scale) = value.get::<f64>() {
126                    self.scale.set(ScaleFactor::new(scale));
127                }
128            }
129            prop::NATURAL_CHILD_SIZE => {
130                if let Ok(enabled) = value.get::<bool>() {
131                    self.natural_child_size.set(enabled);
132                }
133            }
134            prop::MEASURE_ROTATED_BOUNDS => {
135                if let Ok(enabled) = value.get::<bool>() {
136                    self.measure_rotated_bounds.set(enabled);
137                }
138            }
139            prop::FILL => {
140                if let Ok(enabled) = value.get::<bool>() {
141                    self.fill.set(enabled);
142                }
143            }
144            prop::FILL_STRATEGY => {
145                if let Ok(strategy) = value.get::<FillStrategy>() {
146                    self.fill_strategy.set(strategy);
147                }
148            }
149            name => warn!(%name, "unknown RotatedLayout property"),
150        }
151    }

154impl LayoutManagerImpl for RotatedLayoutImpl {
155    fn allocate(&self, widget: &gtk4::Widget, width: i32, height: i32, baseline: i32) {
156        let Some(c) = widget.first_child().filter(|c| c.should_layout()) else {
157            return;
158        };
159        let rotation = self.rotation.get();
160        let allocation = Size::new(width, height);
161        let center = allocation.to_f32().half();
162        let (_, child_nat) = c.measure_size();
163        let target_size: Size<i32> = if self.fill.get() {
164            // The child is stretched so its rendered (rotated and
165            // scaled) bounding box covers the allocation. The exact
166            // `unrotated_bounds` solve runs against the
167            // scale-compensated allocation so the zoom of rotation
168            // animations does not shrink the covered area. The solve
169            // only exists while the viewport aspect stays inside the
170            // `[tan θ, cot θ]` band a rotated rect can bound - outside
171            // it the required child aspect degenerates (negative or
172            // non-finite components near `45° + k·90°`) and
173            // `fill_strategy` takes over.
174            let cover_target = allocation.to_f64().map(|v| v / self.scale.get().as_f64());
175            let solved = cover_target.unrotated_bounds(rotation);
176            let exact_valid = solved.x.is_finite() && solved.y.is_finite() && solved.x > 0.0 && solved.y > 0.0;
177            let nat_aspect = child_nat.to_f64().aspect().unwrap_or(AspectRatio::ONE);
178            match self.fill_strategy.get() {
179                // The aspect-clamped cover reproduces the exact solve
180                // wherever its aspect is in bounds, so it stays
181                // continuous across the whole rotation.
182                FillStrategy::Clamp => cover_target.unrotated_cover(fill_aspect(solved, nat_aspect), rotation).map(|v| v.ceil() as i32),
183                FillStrategy::Fit | FillStrategy::Cover if exact_valid => Size::new(solved.x.ceil() as i32, solved.y.ceil() as i32),
184                // `unrotated_fit` on the unscaled allocation mirrors
185                // the non-fill letterbox exactly, including through
186                // the zoom dip.
187                FillStrategy::Fit => allocation.to_f64().unrotated_fit(nat_aspect, rotation).map(|v| v as i32),
188                FillStrategy::Cover => cover_target.unrotated_cover(nat_aspect, rotation).map(|v| v.ceil() as i32),
189            }
190        } else if self.natural_child_size.get() {
191            // The child keeps its measured natural size: `measure`
192            // reports the rotated bounding box, so the allocation
193            // always contains the rotated child.
194            child_nat
195        } else {
196            // The child keeps its natural aspect ratio and is resized
197            // to the largest rectangle whose rotated bounding box fits
198            // inside the allocation. `unrotated_fit` is a one-sided
199            // condition, so it is never singular - the previous exact
200            // `unrotated_bounds` solve forced a bounds-fill aspect
201            // onto the child, letterboxing aspect-fixed content such
202            // as a wide image down to an invisible sliver.
203            let nat_aspect = child_nat.to_f64().aspect().unwrap_or(AspectRatio::ONE);
204            allocation.to_f64().unrotated_fit(nat_aspect, rotation).map(|v| v as i32)
205        };
206
207        let transform: Transform = CenteredTransform::new(center, target_size.to_f32(), rotation, self.scale.get()).into();
208
209        trace!("size {target_size} baseline {baseline} transform {transform:?}");
210        c.allocate(target_size.x, target_size.y, baseline, Some(transform));
211    }
212
213    fn measure(&self, widget: &gtk4::Widget, orientation: Orientation, _for_size: i32) -> (i32, i32, i32, i32) {
214        let rotation = self.rotation.get();
215
216        let Some(child) = widget.first_child() else {
217            debug!("No child");
218            return (0, 0, -1, -1);
219        };
220        let (child_min, child_nat) = child.measure_size();
221        trace!("child_min {child_min} child_nat {child_nat}");
222
223        // `measure_rotated_bounds` requests the rotated bounding box, so
224        // the allocation always contains the rotated child; disabled,
225        // the request stays unrotated and the child overflows its
226        // allocation symmetrically (`allocate` centers either way).
227        let bounds = |size: Size<i32>| {
228            if self.measure_rotated_bounds.get() {
229                let rotated = size.to_f64().rotated_bounds(rotation).to_i32_ceil();
230                Size::new(rotated.x, rotated.y)
231            } else {
232                size
233            }
234        };
235        let (min, nat) = (bounds(child_min), bounds(child_nat));
236        if orientation == Orientation::Horizontal {
237            (min.x, nat.x, -1, -1)
238        } else {
239            (min.y, nat.y, -1, -1)
240        }
241    }
242}

244/// The child aspect for [`FillStrategy::Clamp`]: the exact solve's
245/// aspect `solved.x / solved.y`, clamped to a bounded distortion range
246/// around the child's natural aspect. Covering with the exact aspect
247/// reproduces `unrotated_bounds` wherever the solve is feasible (both
248/// axes tight at the same value), so the resulting allocation is
249/// continuous across the whole rotation. Inside the infeasible band
250/// the solve's aspect sweeps `∞ → 0` and the clamp morphs the child
251/// bar-to-square-to-bar without jumps; a degenerate `NaN` (square
252/// viewport at `45° + k·90°`) falls back to the natural aspect.
253fn fill_aspect(solved: Size<f64>, natural: AspectRatio) -> AspectRatio {
254    /// Bound on how far the fill solve may distort the child's natural
255    /// aspect ratio - `1` keeps it exactly, `4` allows a `4:1` bar.
256    const MAX_DISTORTION: f64 = 4.0;
257    let natural_f64 = natural.as_f64();
258    let raw = solved.signed_aspect();
259    let clamped = if raw.is_nan() {
260        natural_f64
261    } else if !raw.is_finite() {
262        natural_f64 * MAX_DISTORTION
263    } else {
264        raw.abs().clamp(natural_f64 / MAX_DISTORTION, natural_f64 * MAX_DISTORTION)
265    };
266    AspectRatio::new(clamped).unwrap_or(natural)
