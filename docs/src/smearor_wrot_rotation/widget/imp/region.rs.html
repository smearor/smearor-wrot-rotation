13impl RotatedRegionCalculator for RotationWidgetImpl {
14    fn calculate_rotated_region_with_scale(&self, win_w: f32, win_h: f32, c_w: f32, c_h: f32, deg: f32) -> Region {
15        let rad = deg * PI / 180.0;
16        let (sin, cos) = rad.sin_cos();
17        let _scale_x = win_w / c_w;
18        let _scale_y = win_h / c_h;
19        let pts = [
20            (-win_w / 2.0, -win_h / 2.0),
21            (win_w / 2.0, -win_h / 2.0),
22            (win_w / 2.0, win_h / 2.0),
23            (-win_w / 2.0, win_h / 2.0),
24        ];
25        let mut transformed = Vec::new();
26        for (x, y) in pts {
27            let rx = x * cos - y * sin + win_w / 2.0;
28            let ry = x * sin + y * cos + win_h / 2.0;
29            transformed.push((rx, ry));
30        }
31        let min_y = transformed.iter().map(|(_, y)| *y).fold(f32::INFINITY, f32::min).floor() as i32;
32        let max_y = transformed.iter().map(|(_, y)| *y).fold(f32::NEG_INFINITY, f32::max).ceil() as i32;
33        let region = gtk4::cairo::Region::create();
34        let _step = 1;
35        for y in min_y..max_y {
36            let current_y = y as f32 + 0.5; // Middle of the line
37            let mut intersections = Vec::new();
38            for i in 0..4 {
39                let p1 = transformed[i];
40                let p2 = transformed[(i + 1) % 4];
41                if (p1.1 <= current_y && p2.1 > current_y) || (p2.1 <= current_y && p1.1 > current_y) {
42                    let x = p1.0 + (current_y - p1.1) * (p2.0 - p1.0) / (p2.1 - p1.1);
43                    intersections.push(x);
44                }
45            }
46
47            if intersections.len() >= 2 {
48                intersections.sort_by(|a, b| a.partial_cmp(b).unwrap_or(Ordering::Equal));
49                let start_x = intersections[0];
50                let end_x = intersections[intersections.len() - 1];
51
52                let rect = RectangleInt::new(start_x.round() as i32, y, (end_x - start_x).round() as i32, 1);
53                if let Err(e) = region.union_rectangle(&rect) {
54                    error!("Failed to union rectangle region: {}", e);
55                }
56            }
57        }
58
59        region
60    }
